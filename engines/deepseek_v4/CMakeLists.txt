    if(_strata_ggml_cuda_interface)
        list(REMOVE_ITEM _strata_ggml_cuda_interface CUDA::cudart_static)
        list(REMOVE_ITEM _strata_ggml_cuda_interface "$<LINK_ONLY:CUDA::cudart_static>")
